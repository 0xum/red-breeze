cmake_minimum_required(VERSION 3.25)
project(papirus_inner_corners LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(BUILD_SHARED_LIBS ON)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins")
set(KDE_INSTALL_PLUGINDIR lib/qt6/plugins)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(ECM REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
find_package(KWin 6.8.3 EXACT REQUIRED)
find_package(KDecoration3 REQUIRED)
find_package(Qt6 REQUIRED COMPONENTS CorePrivate Widgets DBus OpenGL)
set(PAPIRUS_PLUGIN_ID "papirus_inner_corners" CACHE STRING "Plugin target ID; override only for live validation")
kcoreaddons_add_plugin(${PAPIRUS_PLUGIN_ID} SOURCES effect.cpp shader.qrc INSTALL_NAMESPACE "kwin/effects/plugins")
target_link_libraries(${PAPIRUS_PLUGIN_ID} PRIVATE KWin::kwin Qt6::CorePrivate KDecoration3::KDecoration)
set_target_properties(${PAPIRUS_PLUGIN_ID} PROPERTIES PREFIX "")
